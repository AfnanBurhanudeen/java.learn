import java.util.Scanner;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int c = sc.nextInt();
       if(c % 4 == 0 && c % 100 != 0){
           System.out.println("Leap year");
       } else{
           System.out.println("Not a leap year");
       }
    }
       }
